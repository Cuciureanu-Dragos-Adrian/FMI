def comp(lista):
    return -lista[2] / lista[1]


fisier = open('input.txt')
greu = float(fisier.readline())

obiecte = []
idx = 1

for linie in fisier:
    ob = linie.split()
    obiecte.append((idx, float(ob[0]), float(ob[1])))
    idx += 1

obiecte.sort(key=comp)

i=0
castig = 0
while greu > 0 and i<len(obiecte):
    if obiecte[i][1] <= greu:
        print('obiect', obiecte[i][0], ': 100.0 %')
        castig += obiecte[i][2]
        greu -= obiecte[i][1]
    else:
        rap = greu/obiecte[i][1]
        print('obiect', obiecte[i][0], ':', round(rap*100, 1) , '%')
        castig += obiecte[i][2] * rap
        greu = 0
    i += 1

print ('catig=', castig)
